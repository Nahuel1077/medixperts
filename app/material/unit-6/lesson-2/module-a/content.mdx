<Title
title="2. Common Terminology"
dictionary={[
    {
    element:"Follow-up",
    explanation:"A future appointment or evaluation with a healthcare provider.",
    extra:"Example: 'Follow up with your primary care physician in one week.'",
    },
    {
    element:"Referral",
    explanation:"A recommendation or authorization for the patient to see another healthcare provider or specialist.",
    extra:"Example: 'You have been referred to a cardiologist.'",
    },
    {
    element:"Wound",
    explanation:"An injury or damaged area of the skin or underlying tissue.",
    extra:"Example: 'Keep the wound clean and dry.'",
    },
    {
    element:"Dressing",
    explanation:"Material placed over a wound to protect it and promote healing.",
    extra:"Example: 'Change the dressing once daily.'",
    },
    {
    element:"Incision",
    explanation:"A surgical cut made in the body during a procedure.",
    extra:"Example: 'Monitor the incision for signs of infection.'",
    },
    {
    element:"Swelling",
    explanation:"An enlargement of a body part caused by fluid accumulation, inflammation, or injury.",
    extra:"Example: 'Some swelling is expected after surgery.'",
    },
    {
    element:"Infection",
    explanation:"The invasion and multiplication of microorganisms that can cause illness.",
    extra:"Example: 'Contact your doctor if you develop signs of infection.'",
    },
    {
    element:"Fever",
    explanation:"An abnormally elevated body temperature, often associated with infection or inflammation.",
    extra:"Example: 'Call your provider if you develop a fever above 101°F.'",
    }, 
    {
    element:"Shortness of breath",
    explanation:"Difficulty breathing or a sensation of not getting enough air.",
    extra:"Example: 'Seek emergency care if you experience severe shortness of breath.'",
    },
    {
    element:"Dizziness",
    explanation:"A sensation of unsteadiness, lightheadedness, or loss of balance.",
    extra:"Example: 'Sit down if you feel dizzy.'",
    },
    {
    element:"Activity restrictions",
    explanation:"Instructions limiting certain physical activities while the patient recovers.",
    extra:"Example: 'Avoid strenuous activity for two weeks.'",
    },
    {
    element:"As tolerated",
    explanation:"An instruction meaning that the patient may perform an activity to the extent that they can comfortably tolerate it.",
    extra:"Example: 'Resume normal activities as tolerated.'",

    ]}
